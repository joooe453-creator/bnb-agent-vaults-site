    if (value >= 1e3) return `$${(value / 1e3).toFixed(value >= 1e5 ? 0 : 1)}K`;

    .then((response) => {
      if (!response.ok) throw new Error(`Registry request failed (${response.status})`);
      return response.json();

    .then((registry) => {
      const liquidityValid = Array.isArray(registry.targets)
        && registry.targets.length > 0
        && registry.targets.every((target) => Number.isFinite(target.liquidity?.usd) && target.liquidity.usd > 0);
      const destinationKindsValid = Array.isArray(registry.yieldMarkets)

      if (registry.schema !== "bnb-agent-vaults/phase1-market-registry@1" || registry.chainId !== 56 || !liquidityValid || !destinationKindsValid) {

      return typeMatch && textMatch;

    }));
