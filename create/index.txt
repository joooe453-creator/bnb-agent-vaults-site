  function problems() {
    const out = [];
    if (!S.registry) out.push(["s-1", "Verified market registry is unavailable."]);
    if (!S.selected.size) out.push(["s-1", "Select at least one target."]);
    if (S.registry && [...S.selected.keys()].some((id) => !S.registry.targets.some((target) => target.id === id))) out.push(["s-1", "A selected target is no longer available."]);
    if (S.registry && [...S.selected.keys()].some((id) => {
      const target = S.registry.targets.find((candidate) => candidate.id === id);
      return target?.dynamic && !(target.discovery?.exactPair && target.swap?.path?.length && target.liquidity?.usd > 0);
    })) out.push(["s-1", "A custom token route has not passed the on-chain pool checks."]);
    const total = [...S.selected.values()].reduce((sum, row) => sum + row.weightBps, 0);
    if (total !== 10000) out.push(["s-1", `Basket weights must total 100% (currently ${(total / 100).toFixed(2)}%).`]);
    if ([...S.selected.values()].some((row) => !(row.weightBps > 0))) out.push(["s-1", "Every selected target needs a positive weight."]);
    if (S.registry && num($("#seed").value) > 0) {
      const deploymentBps = S.mode === "lump-sum" ? 9500 : Math.round(num($("#tranche").value) * 100);
      for (const [id, row] of S.selected.entries()) {
        const target = S.registry.targets.find((candidate) => candidate.id === id);
        const initialOrderUsd = num($("#seed").value) * deploymentBps / 10000 * row.weightBps / 10000;
        const capacityUsd = target?.liquidity?.usd * 0.001;
        if (Number.isFinite(capacityUsd) && initialOrderUsd > capacityUsd) {
          out.push(["s-1", `${displayName(target)} initial order exceeds the 0.1% pool-TVL limit.`]);
        }
      }
    }
    if (S.mode === "accumulate") {
      if (!$("#schedule-on").checked && !$("#dip-on").checked) out.push(["s-2", "Enable a schedule, a drawdown trigger, or both."]);
      if ($("#schedule-on").checked && !(num($("#interval").value) >= 1 && num($("#interval").value) <= 90)) out.push(["s-2", "Schedule must be between 1 and 90 days."]);
      if ($("#dip-on").checked && !(num($("#dip").value) >= 1 && num($("#dip").value) <= 30)) out.push(["s-2", "Drawdown trigger must be between 1% and 30%."]);
      if (!(num($("#tranche").value) >= 1 && num($("#tranche").value) <= 25)) out.push(["s-2", "Tranche size must be between 1% and 25%."]);
    }
    if (!$("#name").value.trim()) out.push(["s-3", "Name the vault."]);
    if (!(num($("#pf").value) >= 0 && num($("#pf").value) <= 30)) out.push(["s-3", "Performance fee must be 0–30%."]);
    if (!(num($("#mf").value) >= 0 && num($("#mf").value) <= 5)) out.push(["s-3", "Management fee must be 0–5%."]);
    if (!(num($("#seed").value) >= 100)) out.push(["s-3", "Seed at least 100 USDT."]);
    return out;
  }
