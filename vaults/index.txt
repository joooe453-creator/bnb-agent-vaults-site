(function () {
  const { $, $$, VAULTS, fmt, perf, riskMeter, statusTag, historyTag, navSeries, sparkline, managerLabel, rulesLabel, vaultRisk, vaultVenues } = BAV;
  BAV.shell("vaults.html");

  const st = { asset: "USDT", filter: "all", q: "", period: "30D", mode: "return", sort: "ra" };

  const tot = VAULTS.reduce((a, v) => a + v.tvl, 0);
  const dep = VAULTS.reduce((a, v) => a + v.followers, 0);
  const agents = new Set(VAULTS.map((v) => v.manager)).size;
  $("#summary").innerHTML = `
    <div><div class="label">Aggregate AUM</div><div class="v">${fmt.usd(tot)}</div></div>
    <div><div class="label">Vaults</div><div class="v">${VAULTS.length}</div></div>
    <div><div class="label">Managers</div><div class="v">${agents}</div></div>
    <div><div class="label">Depositors</div><div class="v">${fmt.n(dep)}</div></div>`;

  function seg(id, key) {
    $$(`#${id} button`).forEach((b) => b.addEventListener("click", () => {
      $$(`#${id} button`).forEach((x) => x.classList.toggle("on", x === b));
      st[key] = b.dataset.v;
      render();
    }));
  }
  seg("f-asset", "asset"); seg("f-filter", "filter"); seg("f-period", "period"); seg("f-mode", "mode");
  $("#f-q").addEventListener("input", (e) => { st.q = e.target.value.toLowerCase(); render(); });
  $("#f-sort").addEventListener("change", (e) => { st.sort = e.target.value; render(); });

  function sliceSeries(v) {
    const s = navSeries(v);
    const d = st.period === "ALL" ? s.length : Math.min(s.length, (st.period === "7D" ? 7 : st.period === "30D" ? 30 : 90) + 1);
    return s.slice(-d);
  }

  function render() {
    const p = st.period, lbl = p === "ALL" ? "Since inception" : p;
    $("#h-perf").textContent = `${lbl} ${st.mode === "annualized" ? "annualized" : "return"}`;
    $("#h-dd").textContent = `${lbl} max DD`;

    let rows = VAULTS.filter((v) => {
      if (v.asset !== st.asset) return false;
      if (st.filter === "new" && v.runtimeDays >= 30) return false;
      if (st.q && !(v.name + " " + v.manager).toLowerCase().includes(st.q)) return false;
      return true;
    });
    const key = {
      ra: (v) => { const r = v.returns[p], d = v.maxDrawdown[p]; return r == null ? -Infinity : r / Math.max(Math.abs(d || 0), 0.25); },
      tvl: (v) => v.tvl,
      perf: (v) => perf(v, p, st.mode) ?? -Infinity,
      dd: (v) => v.maxDrawdown[p] ?? -Infinity,
      age: (v) => v.runtimeDays,
    }[st.sort];
    rows.sort((a, b) => key(b) - key(a));

    $("#dir tbody").innerHTML = rows.map((v) => {
      const r = perf(v, p, st.mode), dd = v.maxDrawdown[p], ht = historyTag(v);
      const copy = v.copyOf ? BAV.vault(v.copyOf.slug) : null;
      const reason = r == null ? (st.mode === "annualized" && v.returns[p] != null ? "Under 30 days" : "Insufficient history") : "";
      return `<tr data-slug="${v.slug}">
        <td><div class="vname"><span class="n">${v.name}</span><span class="m">${managerLabel(v)} · ${v.asset}</span>
          <span class="tags">${statusTag(v)}${ht ? `<span class="tag">${ht}</span>` : ""}${copy ? `<span class="tag warn">Follows ${copy.name} · ${v.copyOf.lag} later</span>` : ""}</span></div></td>
        <td><div style="font-size:14px">${rulesLabel(v)}</div>${riskMeter(vaultRisk(v))}</td>
        <td><span class="logo-stack">${vaultVenues(v).map((id) => BAV.logo(id, "md")).join("")}</span></td>
        <td class="r">${v.sharePrice.toFixed(4)}</td>
        <td class="r ${fmt.cls(r)}">${r == null ? `<span class="na">${reason}</span>` : fmt.pct(r)}</td>
        <td class="r">${dd == null ? `<span class="na">—</span>` : fmt.pct(dd)}</td>
        <td>${sliceSeries(v).length > 2 ? sparkline(sliceSeries(v)) : ""}</td>
        <td class="r">${fmt.usd(v.tvl)}</td>
        <td class="r">${fmt.n(v.followers)}</td>
        <td class="r muted">${v.exitCost}</td>
      </tr>`;
    }).join("") || `<tr><td colspan="10" class="muted" style="padding:40px 0;text-align:center">No vaults match these filters.</td></tr>`;
    $$("#dir tbody tr[data-slug]").forEach((tr) => tr.addEventListener("click", () => (location.href = BAV.route("vault/" + tr.dataset.slug))));
    $("#count").textContent = `Showing ${rows.length} of ${VAULTS.filter((v) => v.asset === st.asset).length} ${st.asset}-currency vaults. Exit cost is today's estimate of the slippage a redeemer pays; it is paid by the person leaving, not by the vault.`;
  }
  render();
})();
